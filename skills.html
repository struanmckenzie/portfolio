    }

        fos = new FileOutputStream(filename);

// Return pointer to user in a hash table or NULL
User *find_user(const HashTable *ht, int user_id)
{
    // validate hash table pointer
    if (ht != NULL)
    {
        // get user's bucket and validate user_id
        int uBucket = get_bucket_for_user(user_id);

        // find user in bucket
        if (uBucket != -1 && ht->buckets[uBucket] != NULL)
        {
            // set current user pointer
            User *current = ht->buckets[uBucket];

            while (current != NULL)
            {
                // if user exists return pointer
                if (current->user_id == user_id)
                    return current;

                // increment current user
                current = current->next;
            }
        }
    }
    // if there are any invalid inputs
    return NULL;
}

std::vector<Edge> AdjacencyMatrix::getNeighbouringEdges(int node) const
{
    // input validation
    if (node > -1 && node < NUMBER_OF_VERTICES) {
        std::vector<Edge> ne;   // store neighbouring edges

        // add neighbouring edges in a loop
        int destNode = 0;
        for (int weight : adjMatrix[node]) {
            // if an edge exists between source and current node
            // add it to neighbouring edges vector
            if (weight > 0) ne.push_back(Edge {node, destNode, weight});
            destNode++;
        }

        // return vector of neighbouring edges
        return ne;
    }

    // if an error occurs
    return {}; // <-- an empty vector
}
